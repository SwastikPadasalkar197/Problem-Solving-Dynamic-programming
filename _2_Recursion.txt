class Fibonacci {
    public static void main(String[] args) {

        int n = 5;
        int a = 0, b = 1, c = 0;

        while (n > 0) {
            System.out.println(a);

            c = a + b;
            a = b;
            b = c;

            n--;
        }
    }
}


class Fibonacci_Numbers_Recursion {

    static int fibonacci(int n) {
        if (n == 0) {
            return 0;
        }

        if (n == 1) {
            return 1;
        }

        return fibonacci(n - 1) + fibonacci(n - 2);
    }

    public static void main(String[] args) {

        int n = 5;

        for (int i = 0; i < n; i++) {
            System.out.println(fibonacci(i));
        }
    }
}
