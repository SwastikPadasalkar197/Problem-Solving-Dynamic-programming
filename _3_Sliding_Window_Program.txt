class Solution {
    public int lengthOfLongestSubstring(String s) {
        int maxLen = 0;

        for (int i = 0; i < s.length(); i++) {
            int len = 0;
            System.out.println("i = " + i);

            for (int j = i; j < s.length(); j++) {
                int k;
                System.out.println("  j = " + j);

                for (k = i; k < j; k++) {
                    System.out.println("    k = " + k);

                    if (s.charAt(k) == s.charAt(j)) {
                        break;
                    }
                }

                if (k != j) {
                    break;
                }

                len++;
            }

            if (len > maxLen) {
                maxLen = len;
            }
        }

        return maxLen;
    }

   
}


/*
Pseudo Code:
Entered a string.

1. Loop 1 to traverse till the end of the string from 1 to the last i.e. n-1.

2. Loop 2 to collect the characters into the framework.

3. Loop 3 to check whether the upcoming character has been succeeding character is duplicated
   or not.

4. Just before the inner loop ends that is loop 3 ends increase the count of the length. as len1.
   -Get out of the inner loop if the duplicate string is found.

5. After breaking out of the 3rd loop. You are now into the 2nd loop. continuing the string    traversal, start comparing adjacent string literals moving forward into the characters. 
   -Repeat step 4 till anothe substring is achieve. we need the count of the len2 

6. Get out of the 3rd loop.

7. Now you are again into the 2nd loop. Compare the len1 and len2.

8. Repeat step 4 to 7 till all the counters reach the last string literal.

9. Atlast, print the string length.

*/
